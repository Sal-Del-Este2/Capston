Esto se uso para encriptar las contraseñas:

-- extensión para encriptar:--
CREATE EXTENSION IF NOT EXISTS pgcrypto;

-- para probar con un usuario --
INSERT INTO usuarios (correo, nickname, nombre, password, rol, estado)
VALUES ('apoderado@correo.cl', 'apoderado', 'Apoderado',
        crypt('ClaveScout#2026', gen_salt('bf')), 'usuario', 'activo');

-- validar al usuario, debe mostrar el hash --
SELECT * FROM usuarios
WHERE correo = 'apoderado@correo.cl'
AND password = crypt('ClaveScout#2026', password);

-- aplicar en toda la tabla --
UPDATE usuarios
SET password = crypt(password, gen_salt('bf'));
