-- consulta global
SELECT 
    u.id AS usuario_id, u.nombre AS usuario_nombre, u.rut, u.telefono, u.nacimiento, u.nickname, u.correo, u.direccion,
    e.id AS estado_id, e.nombre AS estado_nombre,
    r.id AS rol_id, r.nombre AS rol_nombre,
    g.id AS grupo_id, g.nombre AS grupo_nombre,
    c.id AS comuna_id, c.nombre AS comuna_nombre,
    reg.id AS region_id, reg.nombre AS region_nombre
FROM usuarios u
LEFT JOIN estados e ON u.estado_id = e.id
LEFT JOIN roles r ON u.rol_id = r.id
-- Grupo
LEFT JOIN grupos g ON u.grupo_id = g.id
-- Comuna (del usuario)
LEFT JOIN comunas c ON u.comuna_id = c.id
-- Región (desde la comuna del usuario)
LEFT JOIN regiones reg ON c.region_id = reg.id
ORDER BY u.id;
