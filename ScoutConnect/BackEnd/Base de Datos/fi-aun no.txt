CREATE TABLE productos (
    id BIGSERIAL PRIMARY KEY,
    nombre VARCHAR(255),
    descripcion TEXT,
    precio NUMERIC(10,2),
    stock INTEGER
);

CREATE TABLE compras (
    id BIGSERIAL PRIMARY KEY,
    usuario_id BIGINT,
    fecha TIMESTAMP,
    total NUMERIC(10,2)
);

select * from productos order by id;
select * from compras order by id;
